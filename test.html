captureLayer.addEventListener('pointerdown', (e) => { if (!isRecording) return; // ใช้ screenX / screenY จะเริ่มนับ (0, 0) ตั้งแต่ขอบจอดำบนสุดของตัวเครื่องจริง const screenRawX = e.screenX; const screenRawY = e.screenY; // นำมาคูณ DPR ของเครื่อง (เช่น 3.000x) const realX = Math.round(screenRawX * dpr); const realY = Math.round(screenRawY * dpr); // คำนวณเปอร์เซ็นต์เทียบกับขนาดจอจริงทั้งหมด const pctX = ((realX / realScreenWidth) * 100).toFixed(2); const pctY = ((realY / realScreenHeight) * 100).toFixed(2); const pointData = { point: recordedPoints.length + 1, realX: realX, realY: realY, clientX: Math.round(e.clientX), clientY: Math.round(e.clientY), pctX: pctX, pctY: pctY }; recordedPoints.push(pointData); createMarker(e.clientX, e.clientY, pointData.point); if (recordedPoints.length < 4) { updateUIStatus(recordedPoints.length + 1); } else { finishRecording(); } });
